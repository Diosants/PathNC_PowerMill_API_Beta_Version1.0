$featsNew = subtract($extract($featsAfter,'Name'),$extract($featsBefore,'Name'))

FOREACH $ft IN $featsNew {
FOREACH $hol IN components(entity('Featureset',$ft)) {

// Available clolour by name:

// GREEN_YELLOW
// LIGHT_GREY
// MAGENTA
// RED
// RED_MAGENTA
// RED_YELLOW
// WHITE
// YELLOW

IF member($D1, $dia) {
EDIT MODEL ALL COLOUR WHITE
}

IF member($D2, $dia) {
EDIT MODEL ALL COLOUR RED_YELLOW
}

}

}
